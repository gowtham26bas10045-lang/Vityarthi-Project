import mysql.connector

conn = mysql.connector.connect(
    host="localhost",
    user="root",
    password="your_password",
    database="billing_db"
)
cursor = conn.cursor()

cursor.execute("""
CREATE TABLE IF NOT EXISTS customers (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100),
    payment_method VARCHAR(50)
)
""")

cursor.execute("""
CREATE TABLE IF NOT EXISTS items (
    id INT AUTO_INCREMENT PRIMARY KEY,
    customer_id INT,
    item_name VARCHAR(100),
    price FLOAT,
    FOREIGN KEY (customer_id) REFERENCES customers(id)
)
""")

cursor.execute("""
CREATE TABLE IF NOT EXISTS bills (
    id INT AUTO_INCREMENT PRIMARY KEY,
    customer_id INT,
    discount FLOAT,
    gst FLOAT,
    carry_bag_cost FLOAT,
    total_price FLOAT,
    FOREIGN KEY (customer_id) REFERENCES customers(id)
)
""")

conn.commit()

def get_discount(payment_method, total_price):
    if payment_method == 'Cash':
        return 0.05 * total_price
    elif payment_method == 'UPI':
        return 0.10 * total_price
    elif payment_method == 'Card':
        return 0.07 * total_price
    else:
        return 0

def show_invoice(bill_id):
    cursor.execute("""
        SELECT b.id, c.name, c.payment_method, b.discount, b.gst, b.carry_bag_cost, b.total_price
        FROM bills b
        JOIN customers c ON b.customer_id = c.id
        WHERE b.id = %s
    """, (bill_id,))
    bill = cursor.fetchone()

    if not bill:
        print("No invoice found")
        return

    bill_id, name, payment_method, discount, gst, carry_bag_cost, total_price = bill

    cursor.execute("""
        SELECT item_name, price FROM items
        WHERE customer_id = (SELECT customer_id FROM bills WHERE id = %s)
    """, (bill_id,))
    items = cursor.fetchall()

    print("----------------------------------------")
    print("              BILL INVOICE")
    print("----------------------------------------")
    print("Bill ID:", bill_id)
    print("Customer Name:", name)
    print("Payment Method:", payment_method)
    print("Discount:", discount)

    print("\nItems Purchased:")
    for item in items:
        print(item[0], "-", item[1])

    subtotal = total_price - gst - carry_bag_cost
    print("\nSubtotal:", subtotal)
    print("GST:", gst)
    print("Carry Bag Cost:", carry_bag_cost)
    print("Total Amount:", total_price)
    print("----------------------------------------")

def view_past_invoices():
    print("1. View all invoices")
    print("2. Search by Customer Name")
    print("3. Search by Bill ID")
    choice = input("Enter choice: ")

    if choice == "1":
        cursor.execute("""
            SELECT b.id, c.name, b.total_price
            FROM bills b
            JOIN customers c ON b.customer_id = c.id
        """)
        for row in cursor.fetchall():
            print(row)

    elif choice == "2":
        name = input("Enter customer name: ")
        cursor.execute("""
            SELECT b.id, c.name, b.total_price
            FROM bills b
            JOIN customers c ON b.customer_id = c.id
            WHERE c.name LIKE %s
        """, ('%' + name + '%',))
        for row in cursor.fetchall():
            print(row)

    elif choice == "3":
        bill_id = input("Enter bill ID: ")
        show_invoice(bill_id)

while True:
    print("1. Create New Bill")
    print("2. View Past Invoices")
    print("0. Exit")

    main_choice = input("Enter choice: ")

    if main_choice == "1":
        print("1: Cash")
        print("2: UPI")
        print("3: Card")

        name = input("Enter customer name: ")
        payment_choice = int(input("Enter payment method: "))

        if payment_choice == 1:
            payment_method = "Cash"
        elif payment_choice == 2:
            payment_method = "UPI"
        elif payment_choice == 3:
            payment_method = "Card"
        else:
            continue

        cursor.execute(
            "INSERT INTO customers (name, payment_method) VALUES (%s, %s)",
            (name, payment_method)
        )
        conn.commit()
        customer_id = cursor.lastrowid

        num_items = int(input("Enter number of items: "))
        total_price = 0

        for i in range(num_items):
            item_name = input("Enter item name: ")
            item_price = float(input("Enter price: "))
            cursor.execute(
                "INSERT INTO items (customer_id, item_name, price) VALUES (%s, %s, %s)",
                (customer_id, item_name, item_price)
            )
            total_price += item_price

        discount = get_discount(payment_method, total_price)
        carry_bag = input("Carry bag? (y/n): ")
        carry_bag_cost = 10 if carry_bag == 'y' else 0
        gst = 0.18 * (total_price - discount)
        total_with_gst = (total_price - discount) + gst + carry_bag_cost

        cursor.execute("""
            INSERT INTO bills (customer_id, discount, gst, carry_bag_cost, total_price)
            VALUES (%s, %s, %s, %s, %s)
        """, (customer_id, discount, gst, carry_bag_cost, total_with_gst))
        conn.commit()

        bill_id = cursor.lastrowid
        show_invoice(bill_id)

    elif main_choice == "2":
        view_past_invoices()

    elif main_choice == "0":
        break

cursor.close()
conn.close()
